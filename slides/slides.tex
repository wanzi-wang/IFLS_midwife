% =====================================================================
\begin{frame}{Motivation}
  \begin{itemize}[<+->]
    \item Indonesia entered the 1990s with one of Southeast Asia's highest maternal mortality ratios: roughly \alert{450 deaths per 100{,}000 live births} in 1990
    \item The driver: unattended deliveries in rural villages, where the only birth attendants were untrained traditional midwives (\emph{dukun bayi})
    \item The response, \emph{Bidan di Desa} (1989): train and post a certified midwife in each village; by 1996 roughly \alert{54{,}000 midwives} had been placed
    \item A large literature shows early-life health environments shape adult outcomes (Barker 1995; Almond and Currie 2011)
  \end{itemize}
  \vspace{0.6em}
  \onslide<+->{\textbf{Question: does early-life exposure to a village midwife leave measurable effects on human capital a generation later?}}
\end{frame}

% =====================================================================
\begin{frame}{This paper}
  \begin{itemize}[<+->]
    \item \textbf{Data:} Indonesia Family Life Survey (IFLS), all five waves, 1993 to 2014; children born 1984--1999 followed to ages 15--30
    \item \textbf{Treatment:} exposure to the village midwife in utero or at ages 0--3, assigned by the year the midwife first arrived in the community of birth
    \item \textbf{Design:} staggered difference-in-differences; Callaway and Sant'Anna (2021, CS) estimator with never-treated communities as controls
    \item \textbf{Outcomes:} four standardized adult indices, health, cognition, depression (CESD-10), and socioemotional traits (Big Five)
    \item \textbf{Sample:} \NPrimary\ non-migrant individuals across \NVillages\ communities of birth
  \end{itemize}
\end{frame}

% =====================================================================
\begin{frame}{Literature and contribution}
  \muted{\footnotesize \textbf{Early-life environments.} Durable adult effects of early-life interventions (Atwood 2022; Johnson and Jackson 2019; Duflo 2001; Maccini and Yang 2009); an age-two cognitive critical period in Indonesia (Webb 2024).\par}

  \vspace{0.3em}
  \muted{\footnotesize \textbf{This program.} Short-run maternal and child health gains (Frankenberg and Thomas 2001; Frankenberg et al.\ 2005); adult-health effects mostly fade by year ten (Triyana 2016); in-utero exposure improves adult health, cognition, and economic outcomes among men (Ahsan et al.\ 2021).\par}

  \vspace{0.8em}
  \begin{enumerate}[<+->]
    \item \textbf{New outcome margins:} to my knowledge, the first evaluation of this program on adult depression (CESD-10) and Big-Five socioemotional traits
    \item \textbf{Heterogeneity-robust estimation:} prior evaluations rely on pooled TWFE, biased under staggered rollout (Goodman-Bacon 2021); estimator choice reverses the sign of the headline health estimate
  \end{enumerate}
\end{frame}

% =====================================================================
\begin{frame}{Preview of findings}
  \begin{enumerate}[<+->]
    \item Early-life exposure is associated with a \alert{+\CoefHealthCS\ SD} higher adult health index (SE \SEHealthCS) and a \alert{+\CoefCogCS\ SD} higher cognition index (SE \SECogCS), both marginal at the 10\% level
    \item Depression (\CoefDepCS) and socioemotional traits (\CoefSocCS) show no detectable association: the program moves the margins it targeted
    \item \textbf{Mechanism:} in-utero exposure is associated with more \emph{village-midwife-attended} births while aggregate skilled attendance is unchanged, a shift in the \alert{composition} of skilled deliveries, not their volume
  \end{enumerate}
  \vspace{0.6em}
  \onslide<+->{\muted{Estimates are imprecisely pinned down; I read them as suggestive evidence of a positive long-run association.}}
\end{frame}

\section{Program and data}

\begin{frame}{The Village Midwife Program (Bidan di Desa, 1989)}
  Three structural features distinguish the program:
  \begin{enumerate}[<+->]
    \item \textbf{Professional cadre:} three-year nursing diploma with full allied-health certification, not a paraprofessional community-health-worker model
    \item \textbf{Resident provider:} posted to live in the assigned village, with a guaranteed government salary for the first three years
    \item \textbf{Private-practice continuation:} permitted to remain in the village on out-of-pocket payments after the contract
  \end{enumerate}
  \vspace{0.5em}
  \onslide<+->{Services went well beyond delivery care: antenatal visits, postnatal home visits, family planning, immunization referral, iron and vitamin A supplementation \muted{(Makowiecka et al.\ 2008)}: closer to a general early-childhood health intervention than an obstetric-only service}
\end{frame}

\begin{frame}{Data: Indonesia Family Life Survey}
  \begin{itemize}[<+->]
    \item Household panel fielded in 1993, 1997, 2000, 2007, 2014; wave 1 represents 83\% of the population across 13 provinces
    \item \textbf{Rollout year:} Service Availability Roster (SAR), recorded by community informants; an independent PKK module provides an alternative measure used in robustness \hyperlink{app-sarpkk}{\beamergotobutton{SAR vs PKK}}
    \item \textbf{Attrition} is moderate (21\% since wave 1) and uncorrelated with midwife-arrival timing
    \item \textbf{Birthplace:} IFLS does not record community of birth directly; the main sample keeps \alert{non-migrants} (own community identifier unchanged across waves), so the birthplace identifier is clean
    \item Main sample: \NPrimary\ individuals, born 1984--1999, aged 15--30 in 2014, in \NVillages\ communities
  \end{itemize}
\end{frame}

% =====================================================================
\begin{frame}{Outcomes: four adult indices}
  Each domain is an inverse-covariance-weighted index (Anderson 2008), signed so higher is better:
  \vspace{0.3em}
  \begin{itemize}[<+->]
    \item \textbf{Health:} height, not-underweight, lung capacity, no hypertension, early-life health recall, self-rated health
    \item \textbf{Cognition:} IRT-scored number series, serial-7 subtraction, immediate and delayed word recall
    \item \textbf{Depression:} CESD-10 sum, sign-flipped (higher = fewer symptoms)
    \item \textbf{Socioemotional:} Big-Five inventory average (neuroticism sign-flipped)
  \end{itemize}
  \vspace{0.5em}
  \muted{All measured in wave 5 (2014), when the 1984--1999 cohorts are adults aged 15--30.}
\end{frame}

\section{Empirical strategy}

\begin{frame}{Specification}
  Baseline specification for outcome index $Y_{ic}$:
  \begin{equation*}
    Y_{ic} = \beta E_{ic} + X_{ic}'\theta + \eta_c + \gamma_b + \zeta_{w(i)} + \varepsilon_{ic}
  \end{equation*}
  \begin{itemize}[<+->]
    \item $E_{ic} = 1$ if in utero or aged 0--3 at midwife arrival in community $c$ (window from the fetal-origins literature, not estimated)
    \item $\eta_c$, $\gamma_b$, $\zeta_{w(i)}$: community-of-birth, birth-year, and survey-wave fixed effects
    \item $X_{ic}$: sex, mother's schooling, mother's age at birth
    \item Standard errors clustered at the community of birth
  \end{itemize}
  \vspace{0.4em}
  \onslide<+->{\textbf{Identifying assumptions.} Parallel trends; SUTVA violations from midwife mobility plausibly bias toward zero (Ahsan et al.\ 2021)}
\end{frame}

% =====================================================================
\begin{frame}{Estimation under staggered rollout}
  \begin{itemize}[<+->]
    \item Treatment timing is staggered and effects plausibly vary by cohort and event time: exactly the setting where the pooled TWFE coefficient mixes clean and contaminated comparisons with possibly negative weights (Goodman-Bacon 2021)
    \item \textbf{Main estimator:} CS group-time effects $ATT(g,t)$, computed against \alert{never-treated} communities only, then averaged over event times 0--3, so already-treated cohorts never serve as controls
    \item CS requires parallel trends only between each cohort group and the never-treated, with no extrapolation across cohort groups
    \item TWFE is retained as a baseline comparator; de Chaisemartin and D'Haultf\oe uille (dCDH, 2021) as a second heterogeneity-robust estimator
  \end{itemize}
\end{frame}

\begin{frame}{Reading the main results}
  \begin{itemize}[<+->]
    \item \textbf{Health:} +\CoefHealthCS\ SD (SE \SEHealthCS), marginal at the 10\% level; on the order of a \alert{2 cm shift in adult height} (height SD $\approx$ 8 cm carries the largest index weight)
    \item \textbf{Cognition:} +\CoefCogCS\ SD (SE \SECogCS), also marginal at the 10\% level; both magnitudes sit within the range of comparable long-horizon interventions (Duflo 2001; Maccini and Yang 2009; Ahsan et al.\ 2021)
    \item \textbf{Depression and socioemotional:} small, statistically indistinguishable from zero
    \item dCDH agrees in sign and magnitude on health (0.240): the positive estimates reflect the heterogeneity-robust family, not one aggregation rule
    \item \textbf{Estimator choice matters:} pooled TWFE on the same sample gives \alert{$\CoefHealthTWFE$ SD} on health (SE \SEHealthTWFE), the opposite sign, consistent with staggered-rollout bias
  \end{itemize}
\end{frame}

\begin{frame}{Mechanism: composition, not volume}
  \begin{itemize}[<+->]
    \item Only the \alert{village-midwife-specific} attendance margin moves: +4.5 pp on a baseline of 8.3\% ($p\approx0.08$)
    \item Aggregate skilled attendance, prenatal care (near ceiling at 94\%), breastfeeding, and birth weight are all unchanged
    \item Consistent with substitution from other delivery attendants; the flat aggregate keeps the net reallocation within skilled care
    \item These mediators cover the \alert{at-birth margin only}: the postnatal ages 0--3 bundle (home visits, immunization referral, micronutrients) is unmeasured in the pregnancy histories
    \item The first stage is marginal and 10-to-13-year recall may attenuate it; I do not pursue a formal mediation decomposition
  \end{itemize}
\end{frame}

\section{Conclusion}

\begin{frame}{Conclusion}
  \begin{itemize}[<+->]
    \item Children exposed to a village midwife in utero or before age three score \alert{+\CoefHealthCS\ SD} higher on adult health and \alert{+\CoefCogCS\ SD} on cognition a generation later, both marginal at the 10\% level and best read as suggestive
    \item No detectable movement in adult depression or socioemotional traits: clinical maternal-care bundles should not be expected to deliver mental-health gains automatically
    \item The most plausible proximate channel is village-midwife-specific birth attendance, a composition shift within skilled care
    \item \textbf{Policy:} village-level professional cadres may yield modest lasting physical-health and cognition returns, reinforcing the first-thousand-days case; socioemotional margins need directly targeted interventions
  \end{itemize}
\end{frame}
